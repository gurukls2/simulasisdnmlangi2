        // Inisialisasi Tombol Navigasi Grid 1-30
        function initQuestionNavGrid() {
            const grid = document.getElementById('question-nav-grid');
            grid.innerHTML = "";
            for (let i = 0; i < totalQuestions; i++) {
                const btn = document.createElement('button');
                btn.type = "button";
                btn.id = `nav-btn-${i}`;

        // Render Soal ke Layar
        function renderQuestion(index) {
            currentQuestionIndex = index;
            const q = questionsList[index];

            document.getElementById('current-q-number').textContent = q.number;
            document.getElementById('current-q-badge').textContent = q.type;
            document.getElementById('info-question-type').textContent = q.type;

            const area = document.getElementById('question-content-area');
            let htmlOptions = "";
            q.options.forEach(opt => {
                const isSelected = userAnswers[index] === opt.key ? "bg-blue-50 border-blue-600 text-blue-900 font-semibold" : "bg-white border-slate-200 text-slate-800 hover:bg-slate-50";
                const checkedAttr = userAnswers[index] === opt.key ? "checked" : "";
                htmlOptions += `
                    <label onclick="selectAnswer(${index}, '${opt.key}')" class="flex items-start gap-3.5 p-3.5 rounded-xl border-2 ${isSelected} cursor-pointer transition shadow-sm">
                        <input type="radio" name="q_opt_${index}" value="${opt.key}" ${checkedAttr} class="mt-1 text-blue-600 focus:ring-blue-500 h-4 w-4">

            area.innerHTML = `

        // Pilih Jawaban
        function selectAnswer(qIndex, optKey) {
            userAnswers[qIndex] = optKey;
            if (questionStatus[qIndex] !== 'doubt') {
                questionStatus[qIndex] = 'answered';
            }
            renderQuestion(qIndex);
            updateAnsweredCounter();
        }

        // Navigasi Selanjutnya
        function handleNextQuestion() {
            if (currentQuestionIndex < totalQuestions - 1) {
                renderQuestion(currentQuestionIndex + 1);
            }
        }

        // Navigasi Lewati
        function handleSkipQuestion() {
            if (!questionStatus[currentQuestionIndex]) {
                questionStatus[currentQuestionIndex] = 'unanswered';
            }
            handleNextQuestion();
        }

        // Toggle Tombol Ragu-ragu
        function handleToggleDoubt() {
            if (questionStatus[currentQuestionIndex] === 'doubt') {
                questionStatus[currentQuestionIndex] = userAnswers[currentQuestionIndex] ? 'answered' : 'unanswered';
            } else {
                questionStatus[currentQuestionIndex] = 'doubt';
            }
            renderQuestion(currentQuestionIndex);
        }

        // Pindah ke Nomor Tertentu via Grid
        function goToQuestion(index) {
            renderQuestion(index);
        }

        // Update Counter Terjawab
        function updateAnsweredCounter() {
            let answeredCount = 0;
            for (let i = 0; i < totalQuestions; i++) {
                if (userAnswers[i]) answeredCount++;
            }
            document.getElementById('answered-counter').textContent = `${answeredCount} / 30 Terjawab`;
        }

        // Update Tampilan Warna Grid Navigasi
        function updateNavGridUI() {
            for (let i = 0; i < totalQuestions; i++) {
                const btn = document.getElementById(`nav-btn-${i}`);
                if (!btn) continue;

                // Reset kelas dasar
                btn.className = "w-10 h-10 rounded-xl font-bold text-xs border transition flex items-center justify-center cursor-pointer ";

                if (i === currentQuestionIndex) {
                    btn.className += "bg-blue-600 text-white border-blue-700 ring-2 ring-blue-300 shadow-md";
                } else if (questionStatus[i] === 'doubt') {
                    btn.className += "bg-amber-400 text-amber-950 border-amber-500";
                } else if (questionStatus[i] === 'answered' || userAnswers[i]) {
                    btn.className += "bg-emerald-500 text-white border-emerald-600";
                } else {
                    btn.className += "bg-slate-100 text-slate-700 border-slate-200 hover:bg-slate-200";
                }
            }
            updateAnsweredCounter();
        }

        // Timer 75 Menit
        function startTimer() {
            timerInterval = setInterval(() => {
                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    submitAndFinishExam(true); // Waktu habis otomatis submit
                    return;
                }
                timeLeft--;
                const minutes = Math.floor(timeLeft / 60);
                const seconds = timeLeft % 60;
                document.getElementById('timer-text').textContent = 
                    `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
            }, 1000);
        }

        // Hitung Nilai & Kirim ke Google Sheet
        async function submitAndFinishExam(isTimeout = false) {
            if (!isTimeout) {
                document.getElementById('modal-finish-confirm').classList.add('hidden');
            }
            
            // Tampilkan Loading
            document.getElementById('modal-submitting').classList.remove('hidden');
            if (streamGlobal) {
                streamGlobal.getTracks().forEach(track => track.stop());
            }

            // Hitung Skor (Jumlah Benar & Nilai Skala 0-100)
            let correctCount = 0;
            for (let i = 0; i < totalQuestions; i++) {
                if (userAnswers[i] && userAnswers[i] === answerKeys[i]) {
                    correctCount++;
                }
            }
            const finalScore = Math.round((correctCount / totalQuestions) * 100);
            const timestamp = new Date().toLocaleString("id-ID", { timeZone: "Asia/Jakarta" });

            // Payload data yang dikirim ke Google Apps Script Web App
            const payload = {
                timestamp: timestamp,
                nama: studentData.nama,
                kelas: "Kelas 6", // atau sesuaikan jika ada input kelas
                sekolah: "UPT SD Negeri Mlangi 2",
                nilai: finalScore,
                jawabanBenar: correctCount,
                totalSoal: totalQuestions
            };

            try {
                // Mengirim data via fetch POST ke Google Apps Script
                await fetch(SCRIPT_URL, {
                    method: "POST",
                    mode: "no-cors", // Diperlukan untuk Google Apps Script Web App standard CORS policy
                    headers: {
                        "Content-Type": "application/json"
                    },
                    body: JSON.stringify(payload)
                });
            } catch (error) {
                console.error("Gagal mengirim data ke Google Sheet:", error);
            }

            // Tunggu sebentar lalu tampilkan hasil akhir ke peserta
            setTimeout(() => {
                document.getElementById('modal-submitting').classList.add('hidden');
                document.getElementById('page-exam').innerHTML = `
                    <div class="flex-grow flex items-center justify-center p-4 bg-gradient-to-br from-slate-50 to-blue-50">
                        <div class="bg-white rounded-2xl shadow-xl border border-slate-200 p-8 max-w-lg w-full text-center space-y-5 animate-fade-in">
                            <div class="w-20 h-20 bg-emerald-100 text-emerald-600 rounded-full flex items-center justify-center mx-auto text-4xl shadow-inner">
                                <i class="fas fa-check-circle"></i>
                            </div>
                            <h2 class="text-2xl font-bold text-slate-800">Ujian Selesai!</h2>
                            <p class="text-sm text-slate-600">
                                Terima kasih, <strong>${studentData.nama}</strong> (NISN: ${studentData.nisn}), telah menyelesaikan Uji Coba TKA UPT SD Negeri Mlangi 2 Tahun 2027. Data dan nilai Anda telah tersimpan otomatis di database Google Sheet sekolah.
                            </p>
                            <div class="bg-blue-50 border border-blue-200 rounded-xl p-4 grid grid-cols-2 gap-4 text-left">
                                <div>
                                    <span class="text-xs text-slate-500 block">Jawaban Benar:</span>

                                    <span class="text-lg font-bold text-emerald-700">${finalScore}</span>

                                Keluar / Kembali ke Halaman Utama
                            </button>
                        </div>
                    </div>
                `;
            }, 1500);
        }
